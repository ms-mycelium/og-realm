        function switchTab(n) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

            document.getElementById(`content-${n}`).classList.remove('hidden');

                btn.classList.toggle('active', i === n);

            });

        }


        async function getPrice(address) {

            try {

                const res = await fetch(`https://public-api.birdeye.so/public/price?address=${address}`, { headers: { 'X-Chain': 'solana' } });

                const json = await res.json();

                return json.data?.value?.toFixed(6) || '—';

            } catch { return '—'; }

        }


        async function getMarketData(address) {

            try {

                const res = await fetch(`https://public-api.birdeye.so/defi/token_overview?address=${address}`, { headers: { 'X-Chain': 'solana' } });

                const json = await res.json();

                return json.data || {};

            } catch { return {}; }

        }


        async function getPriceHistory(address) {

            const now = Math.floor(Date.now() / 1000);

            const start = now - 86400;

            try {

                const res = await fetch(`https://public-api.birdeye.so/public/history_price?address=${address}&type=15m&time_from=${start}&time_to=${now}`, { headers: { 'X-Chain': 'solana' } });

                const json = await res.json();

                return json.data?.items?.map(i => ({x: new Date(i.unixTime * 1000), y: i.value})) || [];

            } catch { return []; }

        }

                document.getElementById('swap-result').textContent = 'Invalid selection or amount';

                return;

            }

            const inputAmount = Math.floor(amount * Math.pow(10, from.decimals));

            try {

                const res = await fetch(`https://quote-api.jup.ag/v6/quote?inputMint=${from.address}&outputMint=${to.address}&amount=${inputAmount}&slippageBps=50`);

                const quote = await res.json();

                const out = (quote.outAmount / Math.pow(10, to.decimals)).toFixed(6);

                document.getElementById('swap-result').innerHTML = `≈ <span class="font-bold">${out}</span> ${to.symbol}`;

                const execBtn = document.getElementById('execute-swap');

                execBtn.classList.remove('hidden');

                execBtn.onclick = () => window.open(`https://jup.ag/swap/${from.address}-${to.address}?inputAmount=${amount}`, '_blank');

            } catch(e) {

                document.getElementById('swap-result').textContent = 'Quote unavailable';

            }

        });
